@router.post(
    "/register",
    response_model=AuthResponse,
    status_code=status.HTTP_201_CREATED
)
def register(
    request: RegisterRequest,
    db: Session = Depends(get_db)
):
    email = str(request.email).strip().lower()

    existing_user = db.query(User).filter(
        func.lower(User.email) == email
    ).first()

    if existing_user:
        raise HTTPException(
            status_code=status.HTTP_409_CONFLICT,
            detail="An account with this email already exists."
        )

    try:
        user = User(
            full_name=request.full_name.strip(),
            email=email,
            password_hash=hash_password(request.password),
            is_active=True
        )

        organization = Organization(
            name=request.organization_name.strip()
        )

        db.add(user)
        db.add(organization)

        db.flush()

        membership = OrganizationMembership(
            user_id=user.id,
            organization_id=organization.id,
            role="owner"
        )

        db.add(membership)

        db.commit()

        db.refresh(user)
        db.refresh(organization)

    except IntegrityError:
        db.rollback()

        raise HTTPException(
            status_code=status.HTTP_409_CONFLICT,
            detail="Unable to create account. The email may already be registered."
        )

    except Exception:
        db.rollback()
        raise

    access_token = create_access_token(user.id)

    return AuthResponse(
        access_token=access_token,
        user=UserResponse.model_validate(user),
        organization=OrganizationResponse.model_validate(organization)
    )


@router.post(
    "/login",
    response_model=AuthResponse
)
def login(
    request: LoginRequest,
    db: Session = Depends(get_db)
):
    email = str(request.email).strip().lower()

    user = db.query(User).filter(
        func.lower(User.email) == email
    ).first()

    if not user or not verify_password(
        request.password,
        user.password_hash
    ):
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Invalid email or password.",
            headers={"WWW-Authenticate": "Bearer"}
        )

    if not user.is_active:
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="This account has been deactivated."
        )

    membership = db.query(OrganizationMembership).filter(
        OrganizationMembership.user_id == user.id
    ).order_by(OrganizationMembership.created_at.asc()).first()

    if not membership:
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="Your account is not associated with an organization."
        )

    organization = db.query(Organization).filter(
        Organization.id == membership.organization_id
    ).first()

    if not organization:
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="Your organization could not be found."
        )

    access_token = create_access_token(user.id)

    return AuthResponse(
        access_token=access_token,
        user=UserResponse.model_validate(user),
        organization=OrganizationResponse.model_validate(organization)
    )

@router.post("/{document_id}/segment")
def segment_document_clauses(
    document: ContractDocument = Depends(get_authorized_document),
    db: Session = Depends(get_db)
):

    document_id = document.id

    text_units = db.query(DocumentTextUnit).filter(
        DocumentTextUnit.document_id == document_id
    ).order_by(
        DocumentTextUnit.unit_index
    ).all()

    if not text_units:
        raise HTTPException(
            status_code=400,
            detail="Document has not been extracted"
        )

    all_clauses = []

    for unit in text_units:

        clauses = segment_clauses(
            text=unit.extracted_text,
            page_number=unit.page_number
        )

        all_clauses.extend(clauses)

    if not all_clauses:
        raise HTTPException(
            status_code=422,
            detail="No numbered clauses could be identified"
        )

    try:

        db.query(ContractClause).filter(
            ContractClause.document_id == document_id
        ).delete(synchronize_session=False)

        for clause_data in all_clauses:

            clause = ContractClause(
                document_id=document_id,
                **clause_data
            )

            db.add(clause)

        db.commit()

    except Exception:

        db.rollback()

        raise HTTPException(
            status_code=500,
            detail="Clause segmentation failed"
        )

    return {
        "message": "Clauses identified successfully",
        "document_id": str(document_id),
        "total_clauses": len(all_clauses),
        "clauses": [
            {
                "clause_number": clause["clause_number"],
                "clause_title": clause["clause_title"],
                "category": clause["category"],
                "page_number": clause["page_number"]
            }
            for clause in all_clauses
        ]
    }

    "/{contract_id}",
    response_model=ContractResponse
)

    request: ContractUpdateRequest,
    contract: Contract = Depends(
        require_contract_role("owner", "admin")
    ),
    db: Session = Depends(get_db)
):
    updates = request.model_dump(exclude_unset=True)

    if not updates:
        raise HTTPException(
            status_code=400,
            detail="No update fields were provided."
        )

    effective_date = updates.get(
        "effective_date",
        contract.effective_date
    )

    expiration_date = updates.get(
        "expiration_date",
        contract.expiration_date
    )

    if (
        effective_date
        and expiration_date
        and expiration_date < effective_date
    ):
        raise HTTPException(
            status_code=422,
            detail="Expiration date cannot be before effective date."
        )

    for field, value in updates.items():
        setattr(contract, field, value)

    db.commit()
    db.refresh(contract)

    return contract

# 4. Delete document and its physical file
@router.delete(
    "/{document_id}",
    status_code=status.HTTP_204_NO_CONTENT
)
def delete_document(
    document: ContractDocument = Depends(
        require_document_role("owner", "admin")
    ),
    db: Session = Depends(get_db)
):

    upload_root = Path(UPLOAD_DIR).resolve()
    file_path = Path(document.storage_path).resolve()

    # Prevent deletion outside the configured upload directory.
    if (
        file_path == upload_root
        or not file_path.is_relative_to(upload_root)
    ):
        raise HTTPException(
            status_code=400,
            detail="Invalid document storage path."
        )

    quarantine_path = file_path.with_name(
        f"{file_path.name}.deleting-{uuid.uuid4().hex}"
    )

    file_moved = False

    try:
        # Move the file first instead of permanently deleting it.
        if file_path.exists():
            file_path.rename(quarantine_path)
            file_moved = True

        # Delete the database record.
        # Related clauses, evidence and evidence links are handled
        # by the existing database cascade configuration.
        db.delete(document)
        db.commit()

    except Exception:
        db.rollback()

        # Restore the file if the database operation failed.
        if file_moved and quarantine_path.exists():
            quarantine_path.rename(file_path)

        logger.exception(
            "Document deletion failed for document %s",
            document.id
        )

        raise HTTPException(
            status_code=500,
            detail="Document deletion failed."
        )

    # Database deletion succeeded. Permanently remove the quarantined file.
    if file_moved:
        try:
            quarantine_path.unlink(missing_ok=True)
        except OSError:
            logger.exception(
                "Could not remove quarantined file for document %s",
                document.id
            )

    return None

from datetime import datetime, timezone

import logging

router = APIRouter(prefix="/documents", tags=["Document Intelligence"])


@router.post("/{document_id}/process", status_code=status.HTTP_202_ACCEPTED)
def start_document_processing(
    document_id: UUID,
    background_tasks: BackgroundTasks,
    document: ContractDocument = Depends(get_authorized_document),
    db: Session = Depends(get_db),
):
    if document.processing_status == "processing":
        raise HTTPException(
            status_code=status.HTTP_409_CONFLICT,
            detail="Document is already being processed.",
        )

    previous_status = document.processing_status
    previous_started_at = document.processing_started_at
    previous_attempt_id = document.processing_attempt_id
    attempt_id = uuid4()

    try:
        document.processing_status = "processing"
        document.processing_started_at = datetime.now(timezone.utc)
        document.processing_attempt_id = attempt_id
        db.commit()

        background_tasks.add_task(
            process_document_pipeline,
            document_id,
            attempt_id,
        )

    except Exception:
        db.rollback()
        logger.exception(
            "Could not schedule processing for document %s",
            document_id,
        )

        refreshed_document = (
            db.query(ContractDocument)
            .filter(ContractDocument.id == document_id)
            .first()
        )

        if refreshed_document is not None:
            refreshed_document.processing_status = previous_status
            refreshed_document.processing_started_at = previous_started_at
            refreshed_document.processing_attempt_id = previous_attempt_id

            try:
                db.commit()
            except Exception:
                db.rollback()
                logger.exception(
                    "Could not restore processing state for %s",
                    document_id,
                )

        raise HTTPException(
            status_code=status.HTTP_503_SERVICE_UNAVAILABLE,
            detail="Document processing could not be scheduled.",
        )

    return {
        "message": "Document processing started.",
        "document_id": str(document_id),
        "processing_status": "processing",
        "processing_attempt_id": str(attempt_id),
    }

@router.post("/{document_id}/extract-evidence")
def extract_document_evidence(
    document: ContractDocument = Depends(get_authorized_document),
    db: Session = Depends(get_db)
):

    document_id = document.id

    clauses = (
        db.query(ContractClause)
        .filter(
            ContractClause.document_id == document_id
        )
        .order_by(
            ContractClause.clause_number
        )
        .all()
    )

    if not clauses:
        raise HTTPException(
            status_code=422,
            detail="No clauses found. Segment the document first."
        )

    try:

        # Extract everything before changing existing records.
        extracted_records = []

        for clause in clauses:

            extracted = extract_evidence_from_clause(clause)

            for item in extracted:

                extracted_records.append({
                    "clause": clause,
                    "data": item
                })

        # Preserve previous evidence records.
        db.query(EvidenceRecord).filter(
            EvidenceRecord.document_id == document_id,
            EvidenceRecord.is_current.is_(True)
        ).update(
            {"is_current": False},
            synchronize_session=False
        )

        all_evidence = []

        # Insert the new extraction generation.
        for record in extracted_records:

            clause = record["clause"]
            item = record["data"]

            evidence = EvidenceRecord(
                document_id=document_id,
                clause_id=clause.id,
                is_current=True,
                **item
            )

            db.add(evidence)

            all_evidence.append({
                "clause_number": clause.clause_number,
                "claim": item["claim"],
                "evidence_type": item["evidence_type"],
                "page_number": item["page_number"],
            })

        db.commit()

        return {
            "document_id": str(document_id),
            "status": "completed",
            "clauses_processed": len(clauses),
            "evidence_records_created": len(all_evidence),
            "historical_evidence_preserved": True,
            "evidence": all_evidence
        }

    except Exception:
        db.rollback()
        raise

from pathlib import Path
import uuid

from fastapi import APIRouter, Depends, HTTPException

from app.core.database import get_db

from app.models.document import ContractDocument

from app.services.document_parser import extract_document

    prefix="/documents",
    tags=["Document Extraction"]

@router.post("/{document_id}/extract")
def extract_uploaded_document(
    document: ContractDocument = Depends(get_authorized_document),
    db: Session = Depends(get_db)
):

    document_id = document.id

    backend_root = Path(__file__).resolve().parents[3]
    file_path = backend_root / document.storage_path

    if not file_path.is_file():
        raise HTTPException(
            status_code=404,
            detail="Stored document file not found"
        )

    document.processing_status = "processing"
    db.commit()

    try:

        extracted_units = extract_document(
            file_path,
            document.file_type
        )

        nonempty_units = [
            unit for unit in extracted_units
            if unit["text"].strip()
        ]

        if not nonempty_units:

            document.processing_status = "needs_ocr"
            db.commit()

            return {
                "message": "No selectable text found. OCR is required.",
                "status": "needs_ocr"
            }

        db.query(DocumentTextUnit).filter(
            DocumentTextUnit.document_id == document_id
        ).delete(synchronize_session=False)

        for unit in nonempty_units:

            text = unit["text"]

            record = DocumentTextUnit(
                document_id=document_id,
                unit_index=unit["unit_index"],
                page_number=unit["page_number"],
                extracted_text=text,
                character_count=len(text)
            )

            db.add(record)

        document.processing_status = "extracted"

        db.commit()

        return {
            "message": "Document text extracted successfully",
            "document_id": str(document_id),
            "status": "extracted",
            "units_extracted": len(nonempty_units),
            "total_characters": sum(
                len(unit["text"]) for unit in nonempty_units
            )
        }

    except Exception:

        db.rollback()

        document = db.query(ContractDocument).filter(
            ContractDocument.id == document_id
        ).first()

        if document:
            document.processing_status = "failed"
            db.commit()

        raise HTTPException(
            status_code=500,
            detail="Document extraction failed"
        )

def _grouped_counts(db, model, column, *filters):
    rows = (
        db.query(column, func.count(model.id))
        .filter(*filters)
        .group_by(column)
        .all()
    )

    return {

@router.get(
    "/{organization_id}/risk-events",
    response_model=RiskEventTimelineResponse,
)
def get_risk_event_timeline(
    organization_id: UUID,

    contract_id: UUID | None = None,
    risk_finding_id: UUID | None = None,
    event_type: str | None = Query(
        default=None,
        min_length=1,
        max_length=50,
    ),

    page: int = Query(default=1, ge=1),
    page_size: int = Query(default=20, ge=1, le=100),

    membership: OrganizationMembership = Depends(
        require_organization_role("owner", "admin", "member")
    ),

    db: Session = Depends(get_db),
):
    """
    Retrieve the chronological history of risk intelligence events
    belonging to an organization.
    """

    _verify_organization(organization_id, db)

    # Base query is always scoped to the user's organization.
    query = (
        db.query(RiskEvent)
        .join(
            Contract,
            Contract.id == RiskEvent.contract_id,
        )
        .filter(
            Contract.organization_id == organization_id,
        )
    )

    # Optional contract filter.
    if contract_id is not None:
        query = query.filter(
            RiskEvent.contract_id == contract_id,
        )

    # Optional risk finding filter.
    if risk_finding_id is not None:
        query = query.filter(
            RiskEvent.risk_finding_id == risk_finding_id,
        )

    # Optional event type filter.
    if event_type is not None:
        query = query.filter(
            RiskEvent.event_type == event_type.strip(),
        )

    # Calculate total before pagination.
    total = query.count()

    # Calculate pagination metadata.
    total_pages = (
        (total + page_size - 1) // page_size
        if total > 0
        else 0
    )

    offset = (page - 1) * page_size

    # Newest events first, with deterministic tie-breaking.
    events = (
        query
        .order_by(
            RiskEvent.created_at.desc(),
            RiskEvent.id.desc(),
        )
        .offset(offset)
        .limit(page_size)
        .all()
    )

    return RiskEventTimelineResponse(
        organization_id=organization_id,
        total=total,
        page=page,
        page_size=page_size,
        total_pages=total_pages,
        events=events,
    )

def _finding_query(db: Session, organization_id: UUID):
    return (
        db.query(PortfolioRiskFinding)
        .filter(
            PortfolioRiskFinding.organization_id == organization_id
        )
        .options(
            selectinload(
                PortfolioRiskFinding.contract_links
            ).joinedload(
                PortfolioRiskFindingContract.contract
            ),
            selectinload(
                PortfolioRiskFinding.evidence_links
            ).joinedload(
                PortfolioRiskFindingEvidence.evidence_record
            ).joinedload(
                EvidenceRecord.document
            ).joinedload(
                ContractDocument.contract
            ),
        )
    )


def _serialize_finding(finding):
    contracts = []

    for link in finding.contract_links:
        contract = link.contract

        if contract is None:
            continue

        contracts.append({
            "id": contract.id,
            "title": contract.title,
            "status": contract.status,
        })

    evidence = []

    for link in finding.evidence_links:
        record = link.evidence_record

        if record is None:
            continue

        document = record.document
        contract = document.contract if document else None

        evidence.append({
            "evidence_id": record.id,
            "contract_id": contract.id if contract else None,
            "contract_title": contract.title if contract else None,
            "evidence_type": record.evidence_type,
            "claim": record.claim,
            "evidence_text": record.evidence_text,
            "verification_status": record.verification_status,
            "confidence": record.confidence,
            "page_number": record.page_number,
            "clause_id": record.clause_id,
            "document_id": record.document_id,
        })

    return {
        "id": finding.id,
        "organization_id": finding.organization_id,
        "finding_type": finding.finding_type,
        "title": finding.title,
        "description": finding.description,
        "severity": finding.severity,
        "status": finding.status,
        "assessment_status": finding.assessment_status,
        "detection_method": finding.detection_method,
        "rule_id": finding.rule_id,
        "rule_version": finding.rule_version,
        "confidence": finding.confidence,
        "details": finding.details,
        "created_at": finding.created_at,
        "contracts": contracts,
        "evidence": evidence,
    }

@router.post(

    response_model=PortfolioAnalysisResponse,
)
def analyze_portfolio(
    organization_id: UUID,
    membership: OrganizationMembership = Depends(
        require_organization_role("owner", "admin")
    ),
    db: Session = Depends(get_db),
):
    """
    Execute complete organization-level portfolio analysis.

    Pipeline:
        1. Retrieve eligible obligations.
        2. Analyze cross-contract relationships.
        3. Generate potential risk findings.
        4. Persist findings with duplicate protection.
        5. Persist an immutable analysis snapshot.
        6. Return findings requiring human review.

    Only organization owners and administrators can
    trigger portfolio analysis.
    """

    try:
        result = run_organization_portfolio_analysis(
            db=db,
            organization_id=organization_id,
            commit=True,
        )

        finding_ids = [
            finding.id
            for finding in (
                result["created_findings"]
                + result["existing_findings"]
            )
        ]

        findings = []

        if finding_ids:
            persisted_findings = (
                _finding_query(db, organization_id)
                .filter(
                    PortfolioRiskFinding.id.in_(finding_ids)
                )
                .all()
            )

            findings = [
                _serialize_finding(finding)
                for finding in persisted_findings
            ]

        return {
            "organization_id": organization_id,
            "status": "completed",
            "summary": {
                "retrieval": result["retrieval"],
                "intelligence": result["intelligence"],
                "generation": result["generation"],
                "persistence": result["persistence"],
                "snapshot": result.get("snapshot", {}),
            },
            "findings": findings,
            "human_review_queue": result["human_review_queue"],
        }

    except PortfolioFindingValidationError as exc:
        db.rollback()
        raise HTTPException(
            status_code=422,
            detail=str(exc),
        )

    except SQLAlchemyError:
        db.rollback()
        raise HTTPException(
            status_code=500,
            detail="Portfolio analysis could not be completed.",
        )

@router.get(
    "/{organization_id}/portfolio-risk-trends",
    response_model=PortfolioRiskTrendResponse,
)
def get_portfolio_risk_trends(
    organization_id: UUID,
    limit: int = Query(default=10, ge=2, le=50),
    membership: OrganizationMembership = Depends(
        require_organization_role("owner", "admin", "member")
    ),
    db: Session = Depends(get_db),
):
    """
    Analyze historical portfolio risk trends using immutable snapshots.

    The latest `limit` snapshots are loaded for the requested
    organization. This endpoint does not modify snapshots or live findings.
    """

    try:
        snapshots = (
            db.query(PortfolioAnalysisSnapshot)
            .filter(
                PortfolioAnalysisSnapshot.organization_id
                == organization_id
            )
            .options(
                selectinload(PortfolioAnalysisSnapshot.findings)
            )
            .order_by(
                PortfolioAnalysisSnapshot.created_at.desc(),
                PortfolioAnalysisSnapshot.id.desc(),
            )
            .limit(limit)
            .all()
        )

        if len(snapshots) < 2:
            raise HTTPException(
                status_code=409,
                detail=(
                    "At least two portfolio analysis snapshots "
                    "are required to calculate risk trends."
                ),
            )

        return analyze_portfolio_risk_trends(snapshots)

    except PortfolioTrendError as exc:
        raise HTTPException(
            status_code=422,
            detail=str(exc),
        )

    except SQLAlchemyError:
        raise HTTPException(
            status_code=500,
            detail="Portfolio risk trend analysis could not be completed.",
        )
